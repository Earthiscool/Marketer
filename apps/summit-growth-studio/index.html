Autosave is on. The app loads this workspace on page open and saves edits after changes.

Create this table in Supabase SQL Editor:

create table if not exists public.marketing_engine_states (
  id text primary key,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);

alter table public.marketing_engine_states enable row level security;

create policy "Allow shared browser reads"
on public.marketing_engine_states for select
using (true);

create policy "Allow shared browser writes"
on public.marketing_engine_states for insert
with check (true);

create policy "Allow shared browser updates"
on public.marketing_engine_states for update
using (true)
with check (true);
